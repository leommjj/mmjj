    async function getData() {
        let data = [];
        const maps = {};
        const docs = await query(`select * from blocks where tag like '%#Maps#%'`);
        docs.forEach(doc =&gt; {
            const tags = doc.tag.split(' ');
            tags.forEach(tag =&gt; {
                tag = tag.trim().replace(/^#|#$/g, '');
                maps[tag] ? maps[tag]++ : maps[tag] = 1;
            });
        });
        for(const [name, value] of Object .entries(maps)) {
            data.push({name, value});
        }
        return data;

const query = async () =&gt; {
    let dv = Query.DataView(protyle, item, top);
    

    // SQL查询提取城市坐标数据
    const SQL = `
    SELECT content
    FROM blocks 
    WHERE content LIKE '%:: [%' 
    AND content REGEXP '.*:: \\[[0-9.]+,[ ]*[0-9.]+,[ ]*[0-9.]+\\]'
  `;
    
    let blocks = await Query.sql(SQL);
    
    // 处理提取的数据，转换为ECharts需要的格式
    let cityData = [];
    blocks.forEach(block =&gt; {
        // 解析内容格式：城市名:: [经度, 纬度, 数值]
        const match = block.content.match(/(.*):: \[([0-9.]+),\s*([0-9.]+),\s*([0-9.]+)\]/);
        if (match) {
            const cityName = match[1].trim();
            const lng = parseFloat(match[2]);
            const lat = parseFloat(match[3]);
            const value = parseFloat(match[4]);
            
            cityData.push({
                name: cityName,
                value: [lng, lat, value]
            });
        }
    });
    
    // 如果没有找到数据，使用默认数据
    if (cityData.length === 0) {
        cityData = [
            {name: '北京', value: [116.46, 39.92, 2000]},
            {name: '上海', value: [121.48, 31.22, 1800]},
            {name: '杭州', value: [120.19, 30.26, 1000]}
        ];
    }
    
    // 创建ECharts地图配置
    const option = {
        title: {
            text: '城市坐标分布图',
            left: 'center'
        },
        tooltip: {
            trigger: 'item',
            formatter: function(params) {
                if (params.componentType === 'series') {
                    return `${params.name}&lt;br/&gt;坐标: ${params.value[0]}, ${params.value[1]}&lt;br/&gt;数值: ${params.value[2]}`;
                }
                return params.name;
            }
        },
        geo: {
            map: 'china',
            roam: true,
            label: {
                emphasis: {
                    show: false
                }
            },
            itemStyle: {
                normal: {
                    areaColor: '#CDCDCD',
                    borderColor: '#FEFFFF'
                },
                emphasis: {
                    areaColor: '#FEFFFF'
                }
            }
        },
        series: [
            {
                name: '城市',
                type: 'scatter',
                coordinateSystem: 'geo',
                data: cityData,
                symbolSize: function(val) {
                    return Math.sqrt(val[2]) / 5;
                },
                label: {
                    normal: {
                        show: true,
                        formatter: '{b}'
                    }
                },
                itemStyle: {
                    normal: {
                        color: '#FFD983'
                    }
                }
            }
        ]
    };
    
    // 添加ECharts图表
    dv.addecharts(option);
    dv.render();
}

return query();
